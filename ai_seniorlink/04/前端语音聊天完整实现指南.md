## 步骤1: 建立 WebSocket 连接

### 基础代码

```javascript
// 1. 创建 WebSocket 连接
const ws = new WebSocket('ws://192.168.0.103:8000/ws/voice-chat');

// 2. 连接成功回调
ws.onopen = () => {
    console.log('✅ WebSocket 连接已建立');
    // 连接成功后可以开始录音
};

// 3. 接收消息回调
ws.onmessage = (event) => {
    if (event.data instanceof Blob) {
        // 收到音频数据
        console.log('收到音频数据，大小:', event.data.size, 'bytes');
        // 这里稍后会添加播放代码
    } else {
        // 收到文本消息（可能是错误信息）
        console.log('收到文本消息:', event.data);
        try {
            const data = JSON.parse(event.data);
            if (data.error) {
                console.error('服务器错误:', data.error);
            }
        } catch (e) {
            // 不是 JSON，忽略
        }
    }
};

// 4. 错误处理
ws.onerror = (error) => {
    console.error('WebSocket 错误:', error);
};

// 5. 连接关闭回调
ws.onclose = (event) => {
    console.log('连接已关闭:', event.code, event.reason);
    // 可以在这里尝试重连
};
```

## 步骤2: 获取麦克风权限

### 请求麦克风权限

```javascript
let mediaStream = null;

async function requestMicrophonePermission() {
    try {
        // 请求麦克风权限
        mediaStream = await navigator.mediaDevices.getUserMedia({ 
            audio: {
                channelCount: 1,      // 单声道
                sampleRate: 16000,    // 采样率 16kHz（根据豆包API要求调整）
                echoCancellation: true, // 回声消除
                noiseSuppression: true  // 噪声抑制
            } 
        });
        
        console.log('✅ 麦克风权限已获取');
        return true;
    } catch (error) {
        console.error('❌ 无法访问麦克风:', error);
        alert('无法访问麦克风，请检查权限设置');
        return false;
    }
}
```

### 检查浏览器支持

```javascript
// 检查浏览器是否支持

    alert('您的浏览器不支持录音功能');
```

## 步骤3: 开始录音并发送

### 方法1: 使用 MediaRecorder（推荐）

```javascript
let mediaRecorder = null;
let isRecording = false;

async function startRecording() {
    // 1. 检查 WebSocket 连接
    if (ws.readyState !== WebSocket.OPEN) {
        console.error('WebSocket 未连接');
        return false;
    }
    
    // 2. 获取麦克风权限
    if (!mediaStream) {
        const hasPermission = await requestMicrophonePermission();
        if (!hasPermission) {
            return false;
        }
    }
    
    // 3. 创建 MediaRecorder
    try {
        // 选择音频格式（根据豆包API要求）
        const options = {
            mimeType: 'audio/webm;codecs=opus',  // 或 'audio/mp4'
            audioBitsPerSecond: 16000  // 比特率
        };
        
        // 检查浏览器是否支持该格式
        if (!MediaRecorder.isTypeSupported(options.mimeType)) {
            // 如果不支持，使用默认格式
            console.warn('不支持指定格式，使用默认格式');
            mediaRecorder = new MediaRecorder(mediaStream);
        } else {
            mediaRecorder = new MediaRecorder(mediaStream, options);
        }
        
        // 4. 监听数据可用事件
        mediaRecorder.ondataavailable = (event) => {
            if (event.data && event.data.size > 0) {
                // 发送音频数据到服务器
                if (ws.readyState === WebSocket.OPEN) {
                    ws.send(event.data);
                    console.log('发送音频数据块，大小:', event.data.size, 'bytes');
                }
            }
        };
        
        // 5. 监听停止事件
        mediaRecorder.onstop = () => {
            console.log('录音已停止');
            isRecording = false;
        };
        
        // 6. 开始录音（每100ms发送一次数据）
        mediaRecorder.start(100);  // 100ms 间隔
        isRecording = true;
        console.log('✅ 开始录音...');
        
        return true;
    } catch (error) {
        console.error('启动录音失败:', error);
        return false;
    }
}
```

### 方法2: 使用 AudioContext（更底层控制）

```javascript
let audioContext = null;
let mediaStreamSource = null;
let scriptProcessor = null;

async function startRecordingWithAudioContext() {
    // 1. 获取麦克风
    if (!mediaStream) {
        const hasPermission = await requestMicrophonePermission();
        if (!hasPermission) return false;
    }
    
    // 2. 创建 AudioContext
    audioContext = new (window.AudioContext || window.webkitAudioContext)({
        sampleRate: 16000  // 16kHz 采样率
    });
    
    // 3. 创建音频源
    mediaStreamSource = audioContext.createMediaStreamSource(mediaStream);
    
    // 4. 创建 ScriptProcessor（处理音频数据）
    scriptProcessor = audioContext.createScriptProcessor(4096, 1, 1);
    
    // 5. 处理音频数据
    scriptProcessor.onaudioprocess = (event) => {
        if (!isRecording) return;
        
        const inputBuffer = event.inputBuffer;
        const inputData = inputBuffer.getChannelData(0);
        
        // 转换为 16-bit PCM
        const arrayBuffer = new ArrayBuffer(inputData.length * 2);
        const view = new DataView(arrayBuffer);
        
        for (let i = 0; i < inputData.length; i++) {
            const sample = Math.max(-1, Math.min(1, inputData[i]));
            view.setInt16(i * 2, sample < 0 ? sample * 0x8000 : sample * 0x7FFF, true);
        }
        
        // 发送音频数据
        if (ws.readyState === WebSocket.OPEN) {
            ws.send(arrayBuffer);
        }
    };
    
    // 6. 连接节点
    mediaStreamSource.connect(scriptProcessor);
    scriptProcessor.connect(audioContext.destination);
    
    isRecording = true;
    console.log('✅ 开始录音（AudioContext）...');
}
```

```javascript
        handleTextMessage(event.data);
    }

// 播放音频函数
function playAudio(audioBlob) {
    // 方法1: 使用 Audio 元素（简单）
    const audioUrl = URL.createObjectURL(audioBlob);
    const audio = new Audio(audioUrl);
    
    audio.onloadeddata = () => {
        console.log('音频已加载，开始播放');
    };
    
    audio.onended = () => {
        // 播放结束后释放资源
        URL.revokeObjectURL(audioUrl);
    };
    
    audio.onerror = (error) => {
        console.error('播放音频失败:', error);
        URL.revokeObjectURL(audioUrl);
    };
    
    // 播放音频
    audio.play().catch(error => {
        console.error('播放失败:', error);
    });
}

// 使用 AudioContext 播放（更灵活）
let audioContext = null;
let audioQueue = [];  // 音频队列

function initAudioContext() {
    if (!audioContext) {
        audioContext = new (window.AudioContext || window.webkitAudioContext)();
    }
}

async function playAudioWithAudioContext(audioBlob) {
    if (!audioContext) {
        initAudioContext();
    }
    
    try {
        // 将 Blob 转换为 ArrayBuffer
        const arrayBuffer = await audioBlob.arrayBuffer();
        
        // 解码音频数据
        const audioBuffer = await audioContext.decodeAudioData(arrayBuffer);
        
        // 创建音频源
        const source = audioContext.createBufferSource();
        source.buffer = audioBuffer;
        
        // 连接到输出
        source.connect(audioContext.destination);
        
        // 播放
        source.start(0);
        
        console.log('播放音频，时长:', audioBuffer.duration, '秒');
    } catch (error) {
        console.error('播放音频失败:', error);
    }
}
```

## 步骤5: 停止录音和关闭连接

### 停止录音

```javascript
function stopRecording() {
    if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.stop();
        console.log('录音已停止');
    }
    
    if (scriptProcessor) {
        scriptProcessor.disconnect();
        scriptProcessor = null;
    }
    
    if (mediaStreamSource) {
        mediaStreamSource.disconnect();
        mediaStreamSource = null;
    }
    
    isRecording = false;
}

// 释放麦克风资源
function releaseMicrophone() {
    if (mediaStream) {
        mediaStream.getTracks().forEach(track => {
            track.stop();
            console.log('已停止音频轨道');
        });
        mediaStream = null;
    }
}
```

### 关闭连接

```javascript
function closeConnection() {
    // 1. 停止录音
    stopRecording();
    
    // 2. 释放麦克风
    releaseMicrophone();
    
    // 3. 关闭 WebSocket
    if (ws && ws.readyState === WebSocket.OPEN) {
        ws.close();
        console.log('WebSocket 连接已关闭');
    }
}
```

### JavaScript 代码 (voice-chat.js)

```javascript
// 配置
const WS_URL = 'ws://192.168.0.103:8000/ws/voice-chat';

// 全局变量
let ws = null;
let mediaStream = null;
let mediaRecorder = null;
let isRecording = false;
let audioContext = null;

// DOM 元素
const statusEl = document.getElementById('status');
const logContent = document.getElementById('logContent');
const connectBtn = document.getElementById('connectBtn');
const startBtn = document.getElementById('startBtn');
const stopBtn = document.getElementById('stopBtn');
const disconnectBtn = document.getElementById('disconnectBtn');

// 日志函数
function log(message, type = 'info') {
    const time = new Date().toLocaleTimeString();
    const color = type === 'error' ? 'red' : type === 'success' ? 'green' : 'black';
    logContent.innerHTML += `<div style="color: ${color}">[${time}] ${message}</div>`;
    logContent.scrollTop = logContent.scrollHeight;
}

// 更新状态
function updateStatus(connected, recording = false) {
    if (connected) {
        statusEl.textContent = recording ? '已连接 - 录音中' : '已连接';
        statusEl.className = 'status connected';
    } else {
        statusEl.textContent = '未连接';
        statusEl.className = 'status disconnected';
    }
    
    connectBtn.disabled = connected;
    startBtn.disabled = !connected || recording;
    stopBtn.disabled = !recording;
    disconnectBtn.disabled = !connected;
}

// 建立 WebSocket 连接
function connect() {
    log('正在连接 WebSocket...');
    
    ws = new WebSocket(WS_URL);
    
    ws.onopen = () => {
        log('✅ WebSocket 连接成功', 'success');
        updateStatus(true);
    };
    
    ws.onmessage = (event) => {
        if (event.data instanceof Blob) {
            log(`收到音频数据: ${event.data.size} bytes`, 'success');
            playAudio(event.data);
        } else {
            log(`收到文本: ${event.data}`);
            try {
                const data = JSON.parse(event.data);
                if (data.error) {
                    log(`❌ 服务器错误: ${data.error}`, 'error');
                }
            } catch (e) {
                // 不是 JSON
            }
        }
    };
    
    ws.onerror = (error) => {
        log('❌ WebSocket 错误', 'error');
        console.error('WebSocket 错误:', error);
    };
    
    ws.onclose = (event) => {
        log(`连接已关闭: ${event.code} ${event.reason || ''}`);
        updateStatus(false);
        isRecording = false;
    };
}

// 请求麦克风权限
async function requestMicrophonePermission() {
    try {
        mediaStream = await navigator.mediaDevices.getUserMedia({
            audio: {
                channelCount: 1,
                sampleRate: 16000,
                echoCancellation: true,
                noiseSuppression: true
            }
        });
        log('✅ 麦克风权限已获取', 'success');
        return true;
    } catch (error) {
        log(`❌ 无法访问麦克风: ${error.message}`, 'error');
        alert('无法访问麦克风，请检查权限设置');
        return false;
    }
}

// 开始录音
async function startRecording() {
    if (ws.readyState !== WebSocket.OPEN) {
        log('❌ WebSocket 未连接', 'error');
        return;
    }
    
    if (!mediaStream) {
        const hasPermission = await requestMicrophonePermission();
        if (!hasPermission) return;
    }
    
    try {
        const options = {
            mimeType: 'audio/webm;codecs=opus',
            audioBitsPerSecond: 16000
        };
        
        if (!MediaRecorder.isTypeSupported(options.mimeType)) {
            mediaRecorder = new MediaRecorder(mediaStream);
        } else {
            mediaRecorder = new MediaRecorder(mediaStream, options);
        }
        
        mediaRecorder.ondataavailable = (event) => {
            if (event.data && event.data.size > 0) {
                if (ws.readyState === WebSocket.OPEN) {
                    ws.send(event.data);
                }
            }
        };
        
        mediaRecorder.onstop = () => {
            log('录音已停止');
            isRecording = false;
            updateStatus(true, false);
        };
        
        mediaRecorder.start(100);  // 每100ms发送一次
        isRecording = true;
        updateStatus(true, true);
        log('✅ 开始录音...', 'success');
    } catch (error) {
        log(`❌ 启动录音失败: ${error.message}`, 'error');
    }
}

// 停止录音
function stopRecording() {
    if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.stop();
        log('录音已停止');
    }
    isRecording = false;
    updateStatus(true, false);
}

// 播放音频
function playAudio(audioBlob) {
    const audioUrl = URL.createObjectURL(audioBlob);
    const audio = new Audio(audioUrl);
    
    audio.onended = () => {
        URL.revokeObjectURL(audioUrl);
    };
    
    audio.onerror = (error) => {
        log('❌ 播放音频失败', 'error');
        URL.revokeObjectURL(audioUrl);
    };
    
    audio.play().catch(error => {
        log(`❌ 播放失败: ${error.message}`, 'error');
    });
}

// 断开连接
function disconnect() {
    stopRecording();
    
    if (mediaStream) {
        mediaStream.getTracks().forEach(track => track.stop());
        mediaStream = null;
    }
    
    if (ws) {
        ws.close();
        ws = null;
    }
    
    updateStatus(false);
    log('已断开连接');
}

// 页面卸载时清理
window.addEventListener('beforeunload', () => {
    disconnect();
});
```

## 常见问题解决

### 1. 浏览器不支持 getUserMedia

**解决**: 使用 HTTPS 或 localhost（Chrome 要求）

### 2. 音频格式不支持

**解决**: 检查 `MediaRecorder.isTypeSupported()` 并选择支持的格式

### 3. WebSocket 连接失败

**解决**: 
- 检查后端服务是否运行
- 检查防火墙设置
- 检查环境变量配置

### 4. 音频播放延迟

**解决**: 使用 AudioContext 进行更精确的播放控制
