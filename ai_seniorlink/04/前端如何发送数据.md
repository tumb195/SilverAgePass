# 前端如何向后端发送数据

## 🔌 WebSocket 接口地址

```
ws://127.0.0.1:8000/ws/voice-chat
```

---

## 📤 发送数据的方式

后端支持两种类型的数据：

### 1. **文本消息** (Text)
- 用于发送文本指令、配置信息等
- 使用 `ws.send(text)` 发送

### 2. **二进制数据** (Binary)
- 用于发送音频数据
- 使用 `ws.send(audioBuffer)` 发送
- 支持 `ArrayBuffer`、`Blob`、`TypedArray` 等

---

## 💻 完整示例代码

### 基础示例

```javascript
// 1. 建立连接
const ws = new WebSocket('ws://127.0.0.1:8000/ws/voice-chat');

// 2. 连接成功后发送数据
ws.onopen = () => {
    console.log('✅ 连接已建立，可以开始发送数据');
    
    // 发送文本消息
    ws.send('Hello, 这是文本消息');
    
    // 发送音频数据（示例）
    // sendAudioData(audioBuffer);
};

// 3. 发送文本消息
function sendText(text) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(text);  // 直接发送字符串
    }
}

// 4. 发送音频数据
function sendAudio(audioBuffer) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(audioBuffer);  // 发送 ArrayBuffer 或 Blob
    }
}
```

## 🎤 发送音频数据的完整示例

### 方法1: 使用 MediaRecorder（推荐）

```javascript
let mediaRecorder = null;
let audioChunks = [];

// 开始录音并发送
async function startRecordingAndSend() {
    try {
        // 获取麦克风权限
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        
        // 创建 MediaRecorder
        mediaRecorder = new MediaRecorder(stream, {
            mimeType: 'audio/webm;codecs=opus'  // 或 'audio/mp4'
        });
        
        // 监听数据可用事件
        mediaRecorder.ondataavailable = (event) => {
            if (event.data.size > 0) {
                // 直接发送音频数据块
                if (ws.readyState === WebSocket.OPEN) {
                    ws.send(event.data);  // event.data 是 Blob 类型
                }
            }
        };
        
        // 开始录音（每100ms发送一次数据）
        mediaRecorder.start(100);
        console.log('开始录音并发送...');
        
    } catch (error) {
        console.error('无法访问麦克风:', error);
    }
}

// 停止录音
function stopRecording() {
    if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.stop();
        mediaRecorder.stream.getTracks().forEach(track => track.stop());
        console.log('停止录音');
    }
}
```

### 方法2: 使用 AudioContext 采集音频

```javascript
let audioContext = null;
let mediaStreamSource = null;
let scriptProcessor = null;

async function startAudioCapture() {
    try {
        // 获取麦克风
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        
        // 创建 AudioContext
        audioContext = new (window.AudioContext || window.webkitAudioContext)();
        mediaStreamSource = audioContext.createMediaStreamSource(stream);
        
        // 创建 ScriptProcessor（用于处理音频数据）
        scriptProcessor = audioContext.createScriptProcessor(4096, 1, 1);
        
        // 处理音频数据
        scriptProcessor.onaudioprocess = (event) => {
            const inputBuffer = event.inputBuffer;
            const inputData = inputBuffer.getChannelData(0);
            
            // 转换为 ArrayBuffer
            const arrayBuffer = new ArrayBuffer(inputData.length * 2);
            const view = new DataView(arrayBuffer);
            
            for (let i = 0; i < inputData.length; i++) {
                // 转换为 16-bit PCM
                const sample = Math.max(-1, Math.min(1, inputData[i]));
                view.setInt16(i * 2, sample < 0 ? sample * 0x8000 : sample * 0x7FFF, true);
            }
            
            // 发送音频数据
            if (ws.readyState === WebSocket.OPEN) {
                ws.send(arrayBuffer);
            }
        };
        
        // 连接节点
        mediaStreamSource.connect(scriptProcessor);
        scriptProcessor.connect(audioContext.destination);
        
        console.log('开始采集音频...');
        
    } catch (error) {
        console.error('无法访问麦克风:', error);
    }
}

function stopAudioCapture() {
    if (scriptProcessor) {
        scriptProcessor.disconnect();
    }
    if (mediaStreamSource) {
        mediaStreamSource.disconnect();
    }
    if (audioContext) {
        audioContext.close();
    }
}
```

## 🔄 完整工作流程示例

```javascript
class VoiceChatClient {
    constructor(url) {
        this.ws = new WebSocket(url);
        this.mediaRecorder = null;
        this.setupWebSocket();
    }
    
    setupWebSocket() {
        this.ws.onopen = () => {
            console.log('✅ 连接已建立');
            this.onConnected();
        };
        
        this.ws.onmessage = (event) => {
            if (event.data instanceof Blob) {
                this.handleAudioResponse(event.data);
            } else {
                this.handleTextResponse(event.data);
            }
        };
        
        this.ws.onerror = (error) => {
            console.error('WebSocket 错误:', error);
        };
        
        this.ws.onclose = () => {
            console.log('连接已关闭');
        };
    }
    
    // 发送文本消息
    sendText(text) {
        if (this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(text);
        }
    }
    
    // 发送音频数据
    sendAudio(audioData) {
        if (this.ws.readyState === WebSocket.OPEN) {
            this.ws.send(audioData);
        }
    }
    
    // 开始录音并发送
    async startRecording() {
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
            this.mediaRecorder = new MediaRecorder(stream);
            
            this.mediaRecorder.ondataavailable = (event) => {
                if (event.data.size > 0) {
                    this.sendAudio(event.data);
                }
            };
            
            this.mediaRecorder.start(100); // 每100ms发送一次
            console.log('开始录音...');
        } catch (error) {
            console.error('无法访问麦克风:', error);
        }
    }
    
    // 停止录音
    stopRecording() {
        if (this.mediaRecorder) {
            this.mediaRecorder.stop();
            this.mediaRecorder.stream.getTracks().forEach(track => track.stop());
        }
    }
    
    // 处理接收到的音频
    handleAudioResponse(audioBlob) {
        const audio = new Audio(URL.createObjectURL(audioBlob));
        audio.play();
    }
    
    // 处理接收到的文本
    handleTextResponse(text) {
        console.log('收到文本:', text);
    }
    
    onConnected() {
        // 连接成功后的回调
    }
    
    close() {
        this.stopRecording();
        this.ws.close();
    }
}

// 使用示例
const client = new VoiceChatClient('ws://127.0.0.1:8000/ws/voice-chat');

// 连接成功后开始录音
client.ws.onopen = () => {
    client.startRecording();
};

// 停止录音
// client.stopRecording();
```

## 📋 数据格式说明

### 文本消息格式
- **类型**: `string`
- **示例**: `"Hello"`, `'{"action":"start"}'`
- **发送方式**: `ws.send(text)`

### 音频数据格式
- **类型**: `ArrayBuffer` | `Blob` | `TypedArray`
- **常见格式**: 
  - PCM (16-bit)
  - WebM/Opus
  - MP4/AAC
- **发送方式**: `ws.send(audioBuffer)`

---

## ⚠️ 注意事项

1. **连接状态检查**: 发送前检查 `ws.readyState === WebSocket.OPEN`
2. **音频格式**: 确保音频格式符合豆包 API 要求
3. **数据大小**: 建议分块发送，避免单次发送过大的数据
4. **错误处理**: 始终处理连接错误和发送失败的情况
5. **资源清理**: 关闭连接时记得停止录音和释放资源
