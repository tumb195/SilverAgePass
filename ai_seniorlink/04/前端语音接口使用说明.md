# 前端语音聊天接口使用说明

## 🔌 接口地址

### WebSocket 实时语音接口（主要接口）

**本地开发环境**:
```
ws://127.0.0.1:8000/ws/voice-chat
```

**生产环境**（根据实际服务器IP）:
```
ws://<你的服务器IP>:8000/ws/voice-chat
```

例如：`ws://192.168.1.100:8000/ws/voice-chat`

## 📝 前端调用示例

### JavaScript/TypeScript 示例

```javascript
// 1. 建立 WebSocket 连接
const ws = new WebSocket('ws://127.0.0.1:8000/ws/voice-chat');

// 2. 连接成功回调
ws.onopen = () => {
    console.log('✅ 语音连接已建立');
    // 可以开始发送音频数据
};

// 3. 接收消息回调
ws.onmessage = (event) => {
    if (event.data instanceof Blob) {
        // 接收到音频数据（二进制）
        console.log('收到音频数据，大小:', event.data.size);
        
        // 处理音频数据
        handleAudioData(event.data);
    } else {
        // 接收到文本消息
        console.log('收到文本消息:', event.data);
        handleTextMessage(event.data);
    }
};

// 4. 错误处理
ws.onerror = (error) => {
    console.error('WebSocket 错误:', error);
};

// 5. 连接关闭回调
ws.onclose = (event) => {
    console.log('连接已关闭', event.code, event.reason);
};

// 6. 发送音频数据
function sendAudioData(audioBuffer) {
    if (ws.readyState === WebSocket.OPEN) {
        // audioBuffer 可以是 ArrayBuffer、Blob 或 TypedArray
        ws.send(audioBuffer);
    } else {
        console.warn('WebSocket 未连接');
    }
}

// 7. 发送文本消息
function sendTextMessage(text) {
    if (ws.readyState === WebSocket.OPEN) {
        ws.send(text);
    }
}

// 8. 关闭连接
function closeConnection() {
    ws.close();
}

// 处理音频数据的示例函数
function handleAudioData(audioBlob) {
    // 方式1: 创建音频URL并播放
    const audioUrl = URL.createObjectURL(audioBlob);
    const audio = new Audio(audioUrl);
    audio.play();
    
    // 方式2: 使用 AudioContext 处理
    // const audioContext = new AudioContext();
    // audioBlob.arrayBuffer().then(buffer => {
    //     audioContext.decodeAudioData(buffer).then(decodedData => {
    //         // 处理解码后的音频数据
    //     });
    // });
}

// 处理文本消息的示例函数
function handleTextMessage(text) {
    try {
        const data = JSON.parse(text);
        console.log('解析后的消息:', data);
    } catch (e) {
        console.log('文本消息:', text);
    }
}
```

### React Hook 示例

```jsx
import { useEffect, useRef, useState } from 'react';

function VoiceChat() {
    const [isConnected, setIsConnected] = useState(false);
    const [messages, setMessages] = useState([]);
    const wsRef = useRef(null);
    const audioContextRef = useRef(null);

    useEffect(() => {
        // 建立 WebSocket 连接
        const ws = new WebSocket('ws://127.0.0.1:8000/ws/voice-chat');
        wsRef.current = ws;

        ws.onopen = () => {
            console.log('语音连接已建立');
            setIsConnected(true);
        };

        ws.onmessage = (event) => {
            if (event.data instanceof Blob) {
                // 处理音频数据
                handleAudioBlob(event.data);
            } else {
                // 处理文本消息
                setMessages(prev => [...prev, { type: 'text', data: event.data }]);
            }
        };

        ws.onerror = (error) => {
            console.error('WebSocket 错误:', error);
        };

        ws.onclose = () => {
            console.log('连接已关闭');
            setIsConnected(false);
        };

        // 清理函数
        return () => {
            ws.close();
        };
    }, []);

    // 发送音频数据
    const sendAudio = (audioBuffer) => {
        if (wsRef.current && wsRef.current.readyState === WebSocket.OPEN) {
            wsRef.current.send(audioBuffer);
        }
    };

    // 处理接收到的音频
    const handleAudioBlob = (blob) => {
        const audioUrl = URL.createObjectURL(blob);
        const audio = new Audio(audioUrl);
        audio.play();
    };

    return (
        <div>
            <div>连接状态: {isConnected ? '✅ 已连接' : '❌ 未连接'}</div>
            {/* 你的UI组件 */}
        </div>
    );
}
```

### Vue 3 Composition API 示例

```vue
<template>
  <div>
    <div>连接状态: {{ isConnected ? '✅ 已连接' : '❌ 未连接' }}</div>
    <button @click="startRecording">开始录音</button>
    <button @click="stopRecording">停止录音</button>
  </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';

const isConnected = ref(false);
const ws = ref(null);

onMounted(() => {
  // 建立 WebSocket 连接
  ws.value = new WebSocket('ws://127.0.0.1:8000/ws/voice-chat');

  ws.value.onopen = () => {
    console.log('语音连接已建立');
    isConnected.value = true;
  };

  ws.value.onmessage = (event) => {
    if (event.data instanceof Blob) {
      // 处理音频数据
      playAudio(event.data);
    } else {
      console.log('收到文本:', event.data);
    }
  };

  ws.value.onerror = (error) => {
    console.error('WebSocket 错误:', error);
  };

  ws.value.onclose = () => {
    isConnected.value = false;
  };
});

onUnmounted(() => {
  if (ws.value) {
    ws.value.close();
  }
});

const sendAudio = (audioBuffer) => {
  if (ws.value && ws.value.readyState === WebSocket.OPEN) {
    ws.value.send(audioBuffer);
  }
};

const playAudio = (blob) => {
  const audioUrl = URL.createObjectURL(blob);
  const audio = new Audio(audioUrl);
  audio.play();
};
</script>
```

## 🎤 音频采集示例

### 使用 MediaRecorder API

```javascript
let mediaRecorder = null;
let audioChunks = [];

// 开始录音
async function startRecording() {
    try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        mediaRecorder = new MediaRecorder(stream);
        
        mediaRecorder.ondataavailable = (event) => {
            if (event.data.size > 0) {
                // 发送音频数据到 WebSocket
                sendAudioData(event.data);
            }
        };
        
        mediaRecorder.start(100); // 每100ms发送一次数据
        console.log('开始录音');
    } catch (error) {
        console.error('无法访问麦克风:', error);
    }
}

// 停止录音
function stopRecording() {
    if (mediaRecorder && mediaRecorder.state !== 'inactive') {
        mediaRecorder.stop();
        mediaRecorder.stream.getTracks().forEach(track => track.stop());
        console.log('停止录音');
    }
}
```

## 📋 完整使用流程

1. **建立 WebSocket 连接**
   ```javascript
   const ws = new WebSocket('ws://127.0.0.1:8000/ws/voice-chat');
   ```

2. **监听连接状态**
   ```javascript
   ws.onopen = () => { /* 连接成功 */ };
   ws.onmessage = (event) => { /* 接收消息 */ };
   ws.onerror = (error) => { /* 处理错误 */ };
   ws.onclose = () => { /* 连接关闭 */ };
   ```

3. **发送音频数据**
   ```javascript
   ws.send(audioBuffer); // audioBuffer 是 ArrayBuffer 或 Blob
   ```

4. **接收并播放音频**
   ```javascript
   ws.onmessage = (event) => {
       if (event.data instanceof Blob) {
           const audio = new Audio(URL.createObjectURL(event.data));
           audio.play();
       }
   };
   ```

5. **关闭连接**
   ```javascript
   ws.close();
   ```

## ⚠️ 注意事项

1. **协议**: 使用 `ws://`（开发）或 `wss://`（生产，需要HTTPS）
2. **音频格式**: 确保音频格式符合豆包API要求
3. **错误处理**: 务必处理连接错误和断开情况
4. **资源清理**: 关闭连接时记得清理音频资源
5. **浏览器权限**: 需要用户授权麦克风权限
